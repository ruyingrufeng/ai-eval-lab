## 关键命令

```bash
# HyperFrames 渲染
cd benchmarks/video_eval/video-eval-bench
npx hyperframes check          # 质量门禁
npx hyperframes render -q standard --fps 30 -o renders/out.mp4

# Remotion 渲染（复用 HyperFrames Chrome）
cd benchmarks/video_eval/remotion-eval
./node_modules/.bin/remotion render main renders/out.mp4 \
  --browser-executable ~/.cache/hyperframes/chrome/chrome-headless-shell/mac_arm-152.0.7928.2/chrome-headless-shell-mac-arm64/chrome-headless-shell

# FFmpeg 竖版/响度
ffmpeg -i in.mp4 -vf "scale=1080:1920:force_original_aspect_ratio=decrease,pad=1080:1920:(ow-iw)/2:(oh-ih)/2" -c:v libx264 -preset fast out_vertical.mp4
ffmpeg -i in.mp4 -af "loudnorm=I=-16:TP=-1.5:LRA=11" -c:v copy out_loudnorm.mp4
```
